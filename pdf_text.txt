// Soal 4: process all input (continuous keys) 
// Referensi: LearnOpenGL - processInput pattern 
// 
https://learnopengl.com/Getting-started/Transformations 
// 
========================================================
==== 
void processInput(GLFWwindow *window) 
{ 
    if (glfwGetKey(window, GLFW_KEY_ESCAPE) == 
GLFW_PRESS) 
        glfwSetWindowShouldClose(window, true); 
 
    // Translasi - WASD 
    float speed = 0.01f; 
    if (glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS) 
        posY += speed; 
    if (glfwGetKey(window, GLFW_KEY_S) == GLFW_PRESS) 
        posY -= speed; 
    if (glfwGetKey(window, GLFW_KEY_A) == GLFW_PRESS) 
        posX -= speed; 
    if (glfwGetKey(window, GLFW_KEY_D) == GLFW_PRESS) 
        posX += speed; 
 
    // Rotasi - R (kiri) / T (kanan) 
    if (glfwGetKey(window, GLFW_KEY_R) == GLFW_PRESS) 
        rotation += 2.0f; 
    if (glfwGetKey(window, GLFW_KEY_T) == GLFW_PRESS) 
        rotation -= 2.0f; 
 
    // Scale - +/- keys 
    if (glfwGetKey(window, GLFW_KEY_EQUAL) == 
GLFW_PRESS) 
        scaleVal += 0.01f; 
    if (glfwGetKey(window, GLFW_KEY_MINUS) == 
GLFW_PRESS) 
    { 
        scaleVal -= 0.01f; 
        if (scaleVal < 0.05f) 
            scaleVal = 0.05f; 
    } 
} 
 
// 
========================================================
==== 
// Soal 5: key callback untuk event-based input 
(variasi) 

// Referensi: LearnOpenGL - Textures Exercise 4 (UP/DOWN 
mix value) 
// https://learnopengl.com/Getting-started/Textures 
// 
========================================================
==== 
void key_callback(GLFWwindow* window, int key, int 
scancode, int action, int mods) 
{ 
    if (action != GLFW_PRESS) return; 
 
    // Soal 5: UP/DOWN untuk mengubah mix value dari 
texture 
    // Referensi: LearnOpenGL Exercise 4 
    if (key == GLFW_KEY_UP) 
    { 
        mixValue += 0.05f; 
        if (mixValue >= 1.0f) 
            mixValue = 1.0f; 
        std::cout << "Mix Value: " << mixValue << 
std::endl; 
    } 
    if (key == GLFW_KEY_DOWN) 
    { 
        mixValue -= 0.05f; 
        if (mixValue <= 0.0f) 
            mixValue = 0.0f; 
        std::cout << "Mix Value: " << mixValue << 
std::endl; 
    } 
 
    // Soal 5: Set mix value langsung via angka 0-9 
    if (key >= GLFW_KEY_0 && key <= GLFW_KEY_9) 
    { 
        mixValue = (key - GLFW_KEY_0) * 0.1f; 
        std::cout << "Mix Value: " << mixValue << 
std::endl; 
    } 
 
    // Soal 5: Toggle animasi on/off 
    if (key == GLFW_KEY_SPACE) 
    { 
        animationOn = !animationOn; 
        std::cout << "Animasi: " << (animationOn ? "ON" 
: "OFF") << std::endl; 
    } 
 
    // Soal 5: Cycle transform mode 
    if (key == GLFW_KEY_C) 
    { 

        transformMode = (transformMode + 1) % 4; 
        const char* modes[] = {"ALL", "ROTATE ONLY", 
"SCALE ONLY", "TRANSLATE ONLY"}; 
        std::cout << "Transform Mode: " << 
modes[transformMode] << std::endl; 
    } 
 
    // Soal 5: Reset semua ke default 
    if (key == GLFW_KEY_X) 
    { 
        posX = 0.0f; 
        posY = 0.0f; 
        rotation = 0.0f; 
        scaleVal = 0.5f; 
        mixValue = 0.4f; 
        animationOn = true; 
        transformMode = 0; 
        std::cout << "RESET! Mix Value: " << mixValue << 
std::endl; 
    } 
}

// Soal 4: scroll callback untuk zoom (scale) 
// Referensi: LearnOpenGL - scroll_callback 
// 
========================================================

void scroll_callback(GLFWwindow* window, double xoffset, 
double yoffset) 
{ 
    scaleVal += (float)yoffset * 0.05f; 
    if (scaleVal < 0.05f) 
        scaleVal = 0.05f; 
    if (scaleVal > 3.0f) 
        scaleVal = 3.0f; 
}

{ 
    // make sure the viewport matches the new window 
dimensions; note that width and  

    // height will be significantly larger than 
specified on retina displays. 
    glViewport(0, 0, width, height); 
}

2.​ Mix Value Texture 
Source code 
float mixValue = 0.4f;

Penjelasan: 
Pada bagian ini, digunakan variabel mixValue untuk mengatur perbandingan 
antara dua tekstur yang ditampilkan pada objek. Nilai awal ditentukan sebesar 
0.4, yang berasal dari digit terakhir NIM. Nilai ini menunjukkan seberapa 
besar kontribusi masing-masing tekstur dalam proses pencampuran. Ketika 
nilai mendekati 0, tekstur pertama akan lebih dominan, sedangkan jika 
mendekati 1, tekstur kedua akan lebih terlihat. Nilai ini dikirim ke shader 
untuk diproses lebih lanjut sehingga menghasilkan efek blending antara dua 
gambar secara halus.
